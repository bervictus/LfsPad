cmake_minimum_required(VERSION 3.20)
project(LfsPad VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core COMMAND core_tests)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(SDL_DIR "${CMAKE_SOURCE_DIR}/.tools/SDL3-3.4.20" CACHE PATH "SDL source")
  enable_language(C)
  set(SDL_SHARED OFF)
  set(SDL_STATIC ON)
  set(SDL_TEST_LIBRARY OFF)
  set(SDL_TESTS OFF)
  set(SDL_INSTALL OFF)
  foreach(subsystem AUDIO VIDEO RENDER GPU CAMERA SENSOR POWER DIALOG TRAY)
    set(SDL_${subsystem} OFF)
  endforeach()
  # SDL normally probes this header with its audio/video DirectX backends.
  # The Windows LFS target also needs it for older USB joystick adapters.
  set(HAVE_DINPUT_H TRUE)
  add_subdirectory("${SDL_DIR}" sdl EXCLUDE_FROM_ALL)
  set(IMGUI_DIR "${CMAKE_SOURCE_DIR}/.tools/imgui-1.91.9b" CACHE PATH "Dear ImGui source")
  add_library(LfsPad SHARED src/platform.cpp src/runtime.cpp src/controllers.cpp src/insim.cpp src/rumble.cpp src/dinput8.def
    ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(LfsPad PRIVATE src ${IMGUI_DIR} ${IMGUI_DIR}/backends)
  target_compile_definitions(LfsPad PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
    DIRECTINPUT_VERSION=0x0800 LFSPAD_VERSION="${PROJECT_VERSION}")
  target_link_libraries(LfsPad PRIVATE SDL3::SDL3-static d3d11 dxgi user32 gdi32 imm32 bcrypt ws2_32 dxguid)
  if(BUILD_TESTING)
    add_executable(controller_tests tests/controller_tests.cpp src/controllers.cpp)
    target_include_directories(controller_tests PRIVATE src)
    target_compile_definitions(controller_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    target_link_libraries(controller_tests PRIVATE SDL3::SDL3-static dxguid)
    add_test(NAME controllers COMMAND controller_tests)
  endif()
  set_target_properties(LfsPad PROPERTIES PREFIX "" OUTPUT_NAME "dinput8")
  if(MINGW)
    target_link_libraries(LfsPad PRIVATE d3dcompiler_47)
    target_link_options(LfsPad PRIVATE -static-libgcc -static-libstdc++)
  else()
    target_link_libraries(LfsPad PRIVATE d3dcompiler)
  endif()
endif()
